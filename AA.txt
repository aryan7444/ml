def fibonacci_recursive(n):
    if n <= 1:
        return n
    return fibonacci_recursive(n - 1) + fibonacci_recursive(n - 2)

        a, b = b, a + b

def job_sequencing(jobs):

    # Sort jobs according to profit in descending order
    jobs.sort(key=lambda x: x[2], reverse=True)

    # Find maximum deadline
    max_deadline = max(job[1] for job in jobs)

    # Initialize time slots
    slots = [None] * max_deadline

    total_profit = 0

    # Schedule jobs
    for job in jobs:
        job_id, deadline, profit = job

        # Find latest available slot
        for j in range(min(deadline, max_deadline) - 1, -1, -1):

            if slots[j] is None:
                slots[j] = job_id
                total_profit += profit
                break

    # Display scheduled jobs
    print("\nScheduled Jobs:")

    for i in range(max_deadline):
        if slots[i] is not None:
            print(f"Time Slot {i + 1}: {slots[i]}")

    print("\nTotal Profit:", total_profit)

import heapq

# Node class for Huffman Tree
class Node:
    def __init__(self, char, freq):
        self.char = char
        self.freq = freq
        self.left = None
        self.right = None

    def __lt__(self, other):
        return self.freq < other.freq


# Generate Huffman Codes
def generate_codes(root, code, huffman_codes):

    if root is None:
        return

    # Leaf node
    if root.char is not None:
        huffman_codes[root.char] = code if code else "0"
        return

    generate_codes(root.left, code + "0", huffman_codes)
    generate_codes(root.right, code + "1", huffman_codes)


# Huffman Encoding
def huffman_encoding(text):

    if not text:
        print("Input text cannot be empty.")
        return

    # Calculate character frequencies
    frequency = {}

    for char in text:
        frequency[char] = frequency.get(char, 0) + 1

    # Create priority queue
    heap = []

    for char, freq in frequency.items():
        heapq.heappush(heap, Node(char, freq))

    # Build Huffman Tree
    while len(heap) > 1:

        left = heapq.heappop(heap)
        right = heapq.heappop(heap)

        merged = Node(None, left.freq + right.freq)
        merged.left = left
        merged.right = right

        heapq.heappush(heap, merged)

    # Generate Huffman Codes
    root = heapq.heappop(heap)
    huffman_codes = {}

    generate_codes(root, "", huffman_codes)

    # Display results
    print("\nCharacter Frequencies and Huffman Codes:")

    for char, freq in frequency.items():
        display_char = "Space" if char == " " else char
        print(f"{display_char} : Frequency = {freq}, Code = {huffman_codes[char]}")

    # Encode input text
    encoded_text = ""

    for char in text:
        encoded_text += huffman_codes[char]

    print("\nOriginal Text:", text)
    print("Encoded Text:", encoded_text)

    for j in range(1, capacity + 1):

        # If item weight is less than or equal to capacity
        if weight[i - 1] <= j:

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

contract BankAccount {

    address public owner;

    mapping(address => uint256) private balances;

    event Deposited(address indexed user, uint256 amount);
    event Withdrawn(address indexed user, uint256 amount);

    // Constructor
    constructor() {
        owner = msg.sender;
    }

    // 1. Deposit money
    function deposit() public payable {
        require(msg.value > 0, "Deposit must be greater than 0");

        balances[msg.sender] += msg.value;

        emit Deposited(msg.sender, msg.value);
    }

    // 2. Withdraw money
    function withdraw(uint256 amount) public {
        require(amount > 0, "Amount must be greater than 0");
        require(balances[msg.sender] >= amount, "Insufficient balance");

        balances[msg.sender] -= amount;

        (bool success, ) = payable(msg.sender).call{value: amount}("");

        require(success, "Transfer failed");

        emit Withdrawn(msg.sender, amount);
    }

    // 3. Show customer balance
    function showBalance() public view returns (uint256) {
        return balances[msg.sender];
    }

    // 4. Show total contract balance
    function contractBalance() public view returns (uint256) {
        return address(this).balance;
    }
}
